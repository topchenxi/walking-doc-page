一、常见设计模式 ​

1. 单例模式 ​

问题：单例模式

答案： 核心回答：确保类只有一个实例，提供全局访问点。

代码示例：

javascript
class Singleton {
    constructor() {
        if (Singleton.instance) {
            return Singleton.instance;
        }
        Singleton.instance = this;
    }
}

const s1 = new Singleton();
const s2 = new Singleton();
console.log(s1 === s2); // true

// 懒汉式
class LazySingleton {
    static getInstance() {
        if (!LazySingleton.instance) {
            LazySingleton.instance = new LazySingleton();
        }
        return LazySingleton.instance;
    }
}

// 饿汉式
const HungrySingleton = (function() {
    const instance = new Singleton();
    return {
        getInstance: () => instance
    };
})();

2. 观察者模式 ​

问题：观察者模式

答案： 核心回答：对象间一对多依赖，当对象变化时通知所有观察者。

代码示例：

javascript
class Subject {
    constructor() {
        this.observers = [];
    }
    
    attach(observer) {
        this.observers.push(observer);
    }
    
    detach(observer) {
        this.observers = this.observers.filter(o => o !== observer);
    }
    
    notify() {
        this.observers.forEach(observer => observer.update(this));
    }
}

class Observer {
    update(subject) {
        console.log('收到通知:', subject);
    }
}

const subject = new Subject();
const observer1 = new Observer();
const observer2 = new Observer();

subject.attach(observer1);
subject.attach(observer2);
subject.notify();

3. 策略模式 ​

问题：策略模式

答案： 核心回答：定义一系列算法，将每个算法封装起来，使它们可以互换。

代码示例：

javascript
class Strategy {
    execute(data) {
        throw new Error('需要子类实现');
    }
}

class ConcreteStrategyA extends Strategy {
    execute(data) {
        return data.map(x => x * 2);
    }
}

class ConcreteStrategyB extends Strategy {
    execute(data) {
        return data.filter(x => x > 10);
    }
}

class Context {
    constructor(strategy) {
        this.strategy = strategy;
    }
    
    setStrategy(strategy) {
        this.strategy = strategy;
    }
    
    doSomething(data) {
        return this.strategy.execute(data);
    }
}

const context = new Context(new ConcreteStrategyA());
context.doSomething([1, 2, 3, 4]); // [2, 4]
context.setStrategy(new ConcreteStrategyB());
context.doSomething([5, 10, 15]); // [15]

4. 代理模式 ​

问题：代理模式

答案： 核心回答：为其他对象提供代理以控制对这个对象的访问。

代码示例：

javascript
// 虚拟代理
class ImageProxy {
    constructor(url) {
        this.url = url;
        this.realImage = null;
    }
    
    display() {
        if (!this.realImage) {
            this.realImage = new RealImage(this.url);
        }
        this.realImage.display();
    }
}

// 保护代理
class AuthProxy {
    constructor(realService, user) {
        this.realService = realService;
        this.user = user;
    }
    
    request(action) {
        if (!this.user.isAuthenticated()) {
            throw new Error('未授权');
        }
        return this.realService.request(action);
    }
}

// 缓存代理
class CacheProxy {
    constructor() {
        this.cache = new Map();
    }
    
    getData(key) {
        if (this.cache.has(key)) {
            console.log('使用缓存');
            return this.cache.get(key);
        }
        const data = this.fetchData(key);
        this.cache.set(key, data);
        return data;
    }
}

二、前端架构 ​

5. MVC 与 MVVM ​

问题：MVC 与 MVVM 的区别

答案： 核心回答：MVC 中 Controller 处理逻辑，MVVM 中 ViewModel 双向绑定数据。

方面MVCMVVM
数据流单向双向绑定
View展示展示+绑定
Model数据数据+业务
Controller处理逻辑不存在
ViewModel不存在处理逻辑+状态
框架Rails, DjangoVue, Angular

代码示例：

javascript
// MVC
const Model = { data: {} };
const View = {
    render(data) { /* 更新 DOM */ }
};
const Controller = {
    update() {
        const data = Model.data;
        View.render(data);
    }
};

// MVVM (Vue)
new Vue({
    data: { message: 'Hello' }, // Model
    template: '<div>{{ message }}</div>', // View
    methods: {} // ViewModel 逻辑
});

6. 前端组件化 ​

问题：前端组件化

答案： 核心回答：组件化将 UI 拆分为独立可复用的组件。

代码示例：

javascript
// 组件设计原则
// 1. 单一职责
// 2. 通用性
// 3. 可组合性

// 组件规范
const MyComponent = {
    name: 'MyComponent',
    
    props: {
        title: String,
        items: {
            type: Array,
            default: () => []
        }
    },
    
    emits: ['item-click'],
    
    data() {
        return { localData: [] };
    },
    
    computed: {
        computedData() {
            return this.localData.filter(x => x.active);
        }
    },
    
    methods: {
        handleClick(item) {
            this.$emit('item-click', item);
        }
    },
    
    template: `
        <div class="component">
            <h3>{{ title }}</h3>
            <div v-for="item in computedData" @click="handleClick(item)">
                {{ item.name }}
            </div>
        </div>
    `
};

8. 微前端 ​

问题：微前端

答案： 核心回答：微前端将应用拆分为独立可部署的子应用。

代码示例：

javascript
// qiankun 示例
// 主应用
import { registerMicroApps, start } from 'qiankun';

registerMicroApps([
    {
        name: 'vue-app',
        entry: '//localhost:8080',
        container: '#container',
        activeRule: '/vue'
    },
    {
        name: 'react-app',
        entry: '//localhost:8081',
        container: '#container',
        activeRule: '/react'
    }
]);

start();

// 子应用暴露生命周期
export async function bootstrap() {
    console.log('子应用初始化');
}

export async function mount(props) {
    console.log('子应用挂载', props);
}

export async function unmount() {
    console.log('子应用卸载');
}
