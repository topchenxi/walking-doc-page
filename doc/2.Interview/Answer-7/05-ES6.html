// 标签模板
function highlight(strings, ...values) {
  return strings.reduce((prev, cur, i) => prev + cur + (values[i] ? `<strong>${values[i]}</strong>` : ''), '');
}
highlight`Hello, ${name}!`; // "Hello, <strong>Alice</strong>!"

六、Symbol 与迭代器 ​

29-36 Symbol 和 Iterator ​

javascript
const s1 = Symbol('id');  const s2 = Symbol('id');
s1 === s2; // false — 永远唯一

// 自定义迭代器
const iterable = {
  *[Symbol.iterator]() { yield 1; yield 2; yield 3; }
};
for (const v of iterable) { console.log(v); } // 1,2,3

// for...of(值) vs for...in(键)
for (const v of [10, 20]) {} // 10, 20
for (const k in { a: 1 }) {} // "a"

十一、Class 类 ​

66-76 ES6 Class ​

javascript
class Animal {
  #privateField = 'secret';          // 私有字段（ES2022）
  static species = 'animal';         // 静态属性

  constructor(name) { this.name = name; }

  get display() { return this.name; }  // getter
  set display(val) { this.name = val; } // setter

  speak() { return `${this.name} makes a sound`; }
}

class Dog extends Animal {
  constructor(name, breed) {
    super(name);                     // 必须先调 super
    this.breed = breed;
  }
  speak() { return `${super.speak()} and barks`; } // 调用父类方法
}

ES5 vs ES6 继承：

  • ES5：原型链 + 借构造函数
  • ES6：extends，super() 先创建父类实例，内部实现基于原型链但语法更清晰

114-120 Proxy 和 Reflect ​

const proxy = new Proxy(target, {
  get(target, key) { return Reflect.get(target, key); },
  set(target, key, value) { return Reflect.set(target, key, value); },
  has(target, key) { return Reflect.has(target, key); },
  deleteProperty(target, key) { return Reflect.deleteProperty(target, key); }
});
